// AchievementEvents.h
#pragma once
#include <string>
#include <any>
#include <unordered_map>

// 事件类型枚举
enum class EGameEventType
{
    PlayerKill,       // 玩家击杀
    PlayerDeath,      // 玩家死亡
    PlayerLevelUp,    // 玩家升级
    ItemPickup,       // 拾取物品
    QuestComplete,    // 完成任务
    DistanceTraveled, // 累计移动距离
    DamageDealt,      // 造成伤害
};

// 通用游戏事件结构
struct FGameEvent
{
    EGameEventType Type;
    std::unordered_map<std::string, std::any> Params; // 灵活携带参数

    // 便捷构造
    FGameEvent(EGameEventType InType) : Type(InType) {}

    template<typename T>
    FGameEvent& Set(const std::string& Key, T Value)
    {
        Params[Key] = Value;
        return *this;
    }

    template<typename T>
    T Get(const std::string& Key, T Default = T{}) const
    {
        auto It = Params.find(Key);
        if (It != Params.end())
        {
            return std::any_cast<T>(It->second);
        }
        return Default;
    }
};

{

    virtual ~IEventObserver() = default;

    // 核心接口：接收并处理事件

    // 返回该观察者关心哪些事件类型（用于精准订阅）
    virtual std::vector<EGameEventType> GetInterestedEvents() const = 0;
};

// EventBus.h
#pragma once
#include "IEventObserver.h"
#include <unordered_map>
#include <vector>
#include <memory>
#include <algorithm>

class EventBus
{
public:
    // 单例
    static EventBus& Get()
    {
        static EventBus Instance;
        return Instance;
    }

    // 订阅：观察者注册感兴趣的事件
    void Subscribe(IEventObserver* Observer)
    {
        for (EGameEventType EventType : Observer->GetInterestedEvents())
        {
            auto& List = ObserverMap[EventType];
            // 避免重复注册
            if (std::find(List.begin(), List.end(), Observer) == List.end())
            {
                List.push_back(Observer);
            }
        }
    }

    // 取消订阅
    void Unsubscribe(IEventObserver* Observer)
    {
        for (auto& [EventType, List] : ObserverMap)
        {
            List.erase(
                std::remove(List.begin(), List.end(), Observer),
                List.end()
            );
        }
    }

    // 发布事件：通知所有订阅了该事件的观察者
    void Publish(const FGameEvent& Event)
    {
        auto It = ObserverMap.find(Event.Type);
        if (It != ObserverMap.end())
        {
            // 用拷贝遍历，防止回调中修改列表导致迭代器失效
            auto ObserversCopy = It->second;
            for (IEventObserver* Observer : ObserversCopy)
            {
                Observer->OnEvent(Event);
            }
        }
    }

private:
    EventBus() = default;

    // 事件类型 → 观察者列表
    std::unordered_map<EGameEventType, std::vector<IEventObserver*>> ObserverMap;
};

// AchievementDef.h
#pragma once
#include "AchievementEvents.h"
#include <functional>
#include <string>

// 成就解锁条件（函数对象，灵活配置）
using FAchievementCondition = std::function<bool(const FGameEvent& Event, int32_t CurrentProgress)>;

// 成就定义（数据驱动）
struct FAchievementDef
{
    std::string ID;           // 唯一ID，如 "ACH_KILL_100"
    std::string Name;         // "百战之躯"
    std::string Description;  // "累计击杀100名敌人"
    EGameEventType TriggerEvent; // 触发事件类型
    int32_t TargetProgress;   // 目标进度值（如100次击杀）
    FAchievementCondition Condition; // 可选：额外条件（如必须用弓箭击杀）
};

// 成就运行时记录
struct FAchievementRecord
{
    std::string AchievementID;
    int32_t CurrentProgress = 0;
    bool bUnlocked = false;
    std::string UnlockTime;
};

// AchievementManager.h
#pragma once
#include "IEventObserver.h"
#include "AchievementDef.h"
#include <unordered_map>
#include <vector>
#include <iostream>

class AchievementManager : public IEventObserver
{
public:
    AchievementManager()
    {
        RegisterAllAchievements();
        // 自动订阅
        EventBus::Get().Subscribe(this);
    }

    ~AchievementManager()
    {
        EventBus::Get().Unsubscribe(this);
    }

    // ===== IEventObserver 接口实现 =====

    // 声明本系统关心哪些事件（精准订阅，不浪费性能）
    std::vector<EGameEventType> GetInterestedEvents() const override
    {
        return {
            EGameEventType::PlayerKill,
            EGameEventType::PlayerLevelUp,
            EGameEventType::QuestComplete,
            EGameEventType::DistanceTraveled,
        };
    }

    void OnEvent(const FGameEvent& Event) override
    {
        for (auto& Def : AchievementDefs)
        {
            // 只处理与该成就关联的事件
            if (Def.TriggerEvent != Event.Type) continue;

            auto& Record = Records[Def.ID];
            if (Record.bUnlocked) continue; // 已解锁，跳过

            // 检查额外条件
            if (Def.Condition && !Def.Condition(Event, Record.CurrentProgress))
                continue;

            // 更新进度
            Record.CurrentProgress++;

            // 检查是否达成
            if (Record.CurrentProgress >= Def.TargetProgress)
            {
                UnlockAchievement(Def, Record);
            }
            else
            {
                // 通知 UI 更新进度条
                OnProgressUpdated(Def, Record);
            }
        }
    }

private:
    // ===== 成就注册（数据驱动，可从JSON/表格读取）=====
    void RegisterAllAchievements()
    {
        // 成就1：击杀10名敌人
        AchievementDefs.push_back({
            "ACH_KILL_10",
            "初出茅庐",
            "击杀10名敌人",
            EGameEventType::PlayerKill,
            10,
            nullptr // 无额外条件
        });

        // 成就2：击杀100名敌人（用弓箭）
        AchievementDefs.push_back({
            "ACH_BOW_KILL_100",
            "神射手",
            "使用弓箭击杀100名敌人",
            EGameEventType::PlayerKill,
            100,
            // 额外条件：武器类型必须是弓箭
            [](const FGameEvent& Event, int32_t) -> bool {
                return Event.Get<std::string>("WeaponType", "") == "Bow";
            }
        });

        // 成就3：达到10级
        AchievementDefs.push_back({
            "ACH_LEVEL_10",
            "声名鹊起",
            "角色升至10级",
            EGameEventType::PlayerLevelUp,
            1,
            [](const FGameEvent& Event, int32_t) -> bool {
                return Event.Get<int32_t>("NewLevel", 0) >= 10;
            }
        });

        // 初始化所有成就的记录
        for (auto& Def : AchievementDefs)
        {
            Records[Def.ID] = FAchievementRecord{ Def.ID };
        }
    }

    void UnlockAchievement(const FAchievementDef& Def, FAchievementRecord& Record)
    {
        Record.bUnlocked = true;
        Record.UnlockTime = GetCurrentTime();

        // 1. 显示 UI 弹窗
        std::cout << "[🏆 成就解锁] " << Def.Name
                  << " - " << Def.Description << "\n";

        // 2. 存档
        SaveRecord(Record);

        // 3. 可以继续发布"成就解锁"事件，让其他系统响应（如奖励系统）
        FGameEvent AchievementEvent(EGameEventType::QuestComplete);
        AchievementEvent.Set<std::string>("AchievementID", Def.ID);
        EventBus::Get().Publish(AchievementEvent);
    }

    void OnProgressUpdated(const FAchievementDef& Def, const FAchievementRecord& Record)
    {
        std::cout << "[📊 进度更新] " << Def.Name
                  << " (" << Record.CurrentProgress
                  << "/" << Def.TargetProgress << ")\n";
    }

    void SaveRecord(const FAchievementRecord& Record) { /* 写入存档 */ }
    std::string GetCurrentTime() { return "2025-01-01 12:00:00"; }

    std::vector<FAchievementDef> AchievementDefs;
    std::unordered_map<std::string, FAchievementRecord> Records;
};

// GameplaySystem.cpp（业务代码，完全不知道成就系统的存在）

void PlayerCombatSystem::OnEnemyKilled(const std::string& WeaponType)
{
    // 只需发布事件，不关心谁来处理
    FGameEvent Event(EGameEventType::PlayerKill);
    Event.Set<std::string>("WeaponType", WeaponType)
         .Set<int32_t>("EnemyLevel", 5);

    EventBus::Get().Publish(Event);
}

void PlayerSystem::OnLevelUp(int32_t NewLevel)
{
    FGameEvent Event(EGameEventType::PlayerLevelUp);
    Event.Set<int32_t>("NewLevel", NewLevel);

    EventBus::Get().Publish(Event);
}

┌─────────────────────────────────────────────────────────────┐
│                    设计决策说明                              │
├──────────────────┬──────────────────────────────────────────┤
│ 精准订阅         │ Observer 声明感兴趣的事件，避免无效通知   │
│ 条件函数对象     │ Condition 用 std::function，运行时灵活配置│
│ 参数通用化       │ std::any 的 Params Map，事件携参无需改接口│
│ 防迭代器失效     │ Publish 时拷贝观察者列表再遍历            │
│ 完全解耦         │ 业务代码只发事件，不依赖成就系统          │
│ 数据驱动         │ 成就定义可从 JSON/Excel 读取，策划可配置  │
│ 成就可链式触发   │ 解锁成就后再发事件，支持"成就触发成就"    │
└──────────────────┴──────────────────────────────────────────┘

一句话回答面试官： 用 EventBus 作为 Subject 集中管理订阅关系，各子系统（成就、统计、任务）作为 Observer 各自订阅感兴趣的事件；业务层只需 Publish 事件，完全不耦合成就逻辑，新增成就只需注册定义，零改动业务代码。
