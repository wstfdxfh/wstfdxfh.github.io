struct ListNode {
    int val;
    ListNode* next;
    ListNode(int x) : val(x), next(nullptr) {}
};

ListNode* reverseList(ListNode* head) {
    ListNode* prev = nullptr;
    ListNode* curr = head;
    while (curr) {
        ListNode* next = curr->next;   // 1. 暂存后继
        curr->next = prev;             // 2. 反转指向
        prev = curr;                   // 3. prev 前移
        curr = next;                   // 4. curr 前移
    }
    return prev;
}

ListNode* reverseList(ListNode* head) {
    if (!head || !head->next) return head;
    ListNode* newHead = reverseList(head->next);
    head->next->next = head;   // 后继节点指回自己
    head->next = nullptr;      // 断开原有指向
    return newHead;
}

void quickSort(vector<int>& arr, int left, int right) {
    if (left >= right) return;
    // 选取 pivot（可用三数取中法优化）
    int pivot = arr[left];
    int i = left, j = right;
    while (i < j) {
        while (i < j && arr[j] >= pivot) j--;   // 从右找小的
        while (i < j && arr[i] <= pivot) i++;    // 从左找大的
        if (i < j) swap(arr[i], arr[j]);
    }
    swap(arr[left], arr[i]);   // pivot 归位
    quickSort(arr, left, i - 1);
    quickSort(arr, i + 1, right);
}

复杂度： 平均 O(n log n)，最差 O(n²)（已有序时退化）。空间 O(log n) 递归栈。

常见追问：

  • 如何优化最差情况？ 随机选 pivot 或三数取中法。
  • 快排稳定吗？ 不稳定。归并排序是稳定的 O(n log n) 排序。
  • 什么时候用归并？ 链表排序（无随机访问）、需要稳定排序、外部排序（大文件）。

3. 无重复字符的最长子串#

int lengthOfLongestSubstring(string s) {
    vector<int> lastPos(128, -1);   // 字符上次出现的位置
    int left = 0, maxLen = 0;
    for (int right = 0; right < (int)s.size(); right++) {
        if (lastPos[s[right]] >= left)
            left = lastPos[s[right]] + 1;   // 跳到重复字符之后
        lastPos[s[right]] = right;
        maxLen = max(maxLen, right - left + 1);
    }
    return maxLen;
}

复杂度： O(n) 时间，O(字符集大小) 空间。

核心思路： 维护一个窗口 [left, right]，right 右扩，遇到重复字符就将 left 跳到上次出现位置的下一位。窗口内始终无重复。

int binarySearch(vector<int>& nums, int target) {
    int left = 0, right = (int)nums.size() - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;   // 防溢出
        if (nums[mid] == target) return mid;
        else if (nums[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}

int lowerBound(vector<int>& nums, int target) {
    int left = 0, right = (int)nums.size();
    while (left < right) {
        int mid = left + (right - left) / 2;
        if (nums[mid] < target) left = mid + 1;
        else right = mid;
    }
    return left;   // 即插入位置
}

关键易错点： <= vs <、right = mid vs right = mid - 1——边界条件是二分查找的核心。面试中写错边界是最常见的翻车原因。

float 精度追问（米哈游）： float 比较不能用 ==，需要用 fabs(a - b) < epsilon，float 有效精度约 6~7 位小数。

ListNode* mergeTwoLists(ListNode* l1, ListNode* l2) {
    ListNode dummy(0);
    ListNode* tail = &dummy;
    while (l1 && l2) {
        if (l1->val <= l2->val) { tail->next = l1; l1 = l1->next; }
        else                    { tail->next = l2; l2 = l2->next; }
        tail = tail->next;
    }
    tail->next = l1 ? l1 : l2;
    return dummy.next;
}

ListNode* mergeKLists(vector<ListNode*>& lists) {
    auto cmp = [](ListNode* a, ListNode* b) { return a->val > b->val; };
    priority_queue<ListNode*, vector<ListNode*>, decltype(cmp)> pq(cmp);

    for (auto head : lists)
        if (head) pq.push(head);

    ListNode dummy(0);
    ListNode* tail = &dummy;
    while (!pq.empty()) {
        ListNode* node = pq.top(); pq.pop();
        tail->next = node;
        tail = tail->next;
        if (node->next) pq.push(node->next);
    }
    return dummy.next;
}

复杂度： 合并两个 O(m+n)；合并 K 个用堆 O(N log K)，N 是总元素数。

bool hasCycle(ListNode* head) {
    ListNode *slow = head, *fast = head;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) return true;
    }
    return false;
}

ListNode* detectCycle(ListNode* head) {
    ListNode *slow = head, *fast = head;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) {
            fast = head;                  // fast 回到头部
            while (slow != fast) {        // 同速走，再相遇即入口
                slow = slow->next;
                fast = fast->next;
            }
            return slow;
        }
    }
    return nullptr;
}

数学原理： 设头→入口 = a，入口→相遇 = b，相遇→入口 = c。相遇时 slow 走了 a+b，fast 走了 a+b+b+c = 2(a+b)，所以 a = c。

7. 设计 LRU 缓存#

class LRUCache {
    int capacity;
    list<pair<int,int>> cache;                                  // front=最新
    unordered_map<int, list<pair<int,int>>::iterator> map;      // key→迭代器

public:
    LRUCache(int cap) : capacity(cap) {}

    int get(int key) {
        if (!map.count(key)) return -1;
        cache.splice(cache.begin(), cache, map[key]);   // 移到头部
        return map[key]->second;
    }

    void put(int key, int value) {
        if (map.count(key)) {
            map[key]->second = value;
            cache.splice(cache.begin(), cache, map[key]);
        } else {
            if ((int)cache.size() == capacity) {
                map.erase(cache.back().first);          // 淘汰尾部
                cache.pop_back();
            }
            cache.push_front({key, value});
            map[key] = cache.begin();
        }
    }
};

核心： 哈希表 O(1) 查找 + 双向链表 O(1) 增删移动。splice 是 std::list 的 O(1) 节点转移操作。

游戏应用： 资源缓存管理（纹理/模型/音频），当内存不足时淘汰最久未使用的资源。

int partition(vector<int>& nums, int left, int right) {
    int pivot = nums[left + rand() % (right - left + 1)];  // 随机 pivot
    int i = left, j = right;
    while (i <= j) {
        while (nums[i] > pivot) i++;    // 降序
        while (nums[j] < pivot) j--;
        if (i <= j) { swap(nums[i], nums[j]); i++; j--; }
    }
    return i;   // i 是右半部分的起点
}

int findKthLargest(vector<int>& nums, int k) {
    int left = 0, right = (int)nums.size() - 1;
    while (true) {
        int pos = partition(nums, left, right);
        if (pos == k) return nums[pos - 1];
        // 简化版：用 nth_element 思想
        if (k - 1 < pos) right = pos - 1;
        else left = pos;
    }
}

int findKthLargest(vector<int>& nums, int k) {
    priority_queue<int, vector<int>, greater<int>> minHeap;  // 最小堆
    for (int n : nums) {
        minHeap.push(n);
        if ((int)minHeap.size() > k) minHeap.pop();          // 维护大小为 k
    }
    return minHeap.top();
}

复杂度： QuickSelect 平均 O(n)；堆方法 O(n log k)。海量数据场景用堆更稳定。

vector<vector<int>> levelOrder(TreeNode* root) {
    vector<vector<int>> result;
    if (!root) return result;
    queue<TreeNode*> q;
    q.push(root);
    while (!q.empty()) {
        int size = q.size();
        vector<int> level;
        for (int i = 0; i < size; i++) {
            TreeNode* node = q.front(); q.pop();
            level.push_back(node->val);
            if (node->left)  q.push(node->left);
            if (node->right) q.push(node->right);
        }
        result.push_back(level);
    }
    return result;
}

  • 判断完全二叉树： 层序遍历中遇到 null 后，后续不应再有非 null 节点。

int climbStairs(int n) {
    if (n <= 2) return n;
    int a = 1, b = 2;
    for (int i = 3; i <= n; i++) {
        int c = a + b;     // f(n) = f(n-1) + f(n-2)
        a = b;
        b = c;
    }
    return b;
}

int climbStairs_2_3(int n) {
    if (n <= 1) return 0;
    if (n == 2) return 1;
    if (n == 3) return 1;
    vector<int> dp(n + 1, 0);

11. 二叉树的最近公共祖先#

高频指数：★★★★☆ | LeetCode 236 | 来源：腾讯 IEG Top10（出现 2 次）、字节

真实面经： CodeTop 腾讯 IEG 数据；字节面经"二叉树的最近公共祖先，原题 2 个节点，面试变成多个节点"。

解答（递归后序遍历）：

TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
    if (!root || root == p || root == q) return root;
    TreeNode* left  = lowestCommonAncestor(root->left,  p, q);
    TreeNode* right = lowestCommonAncestor(root->right, p, q);
    if (left && right) return root;     // p、q 分布在两侧，root 就是 LCA
    return left ? left : right;         // 都在同一侧
}

核心思路：

  1. 递归到底，找到 p 或 q 就返回
  2. 如果一个节点的左右子树分别返回了非空，说明 p 和 q 分布在两侧，该节点就是 LCA
  3. 如果只有一侧非空，LCA 在那一侧

复杂度： O(n) 时间，O(n) 递归栈空间。

#include <mutex>
class Singleton {
    static Singleton* instance;
    static std::mutex mtx;
    Singleton() {}
public:
    static Singleton* getInstance() {
        if (!instance) {                          // 第一次检查（无锁快速路径）
            std::lock_guard<std::mutex> lock(mtx);
            if (!instance)                        // 第二次检查（防多线程重复创建）
                instance = new Singleton();
        }
        return instance;
    }
};
Singleton* Singleton::instance = nullptr;

float cross2D(Vec2 a, Vec2 b) { return a.x * b.y - a.y * b.x; }

bool pointInTriangle(Vec2 P, Vec2 A, Vec2 B, Vec2 C) {
    float d1 = cross2D(B - A, P - A);
    float d2 = cross2D(C - B, P - B);
    float d3 = cross2D(A - C, P - C);
    bool hasNeg = (d1 < 0) || (d2 < 0) || (d3 < 0);
    bool hasPos = (d1 > 0) || (d2 > 0) || (d3 > 0);
    return !(hasNeg && hasPos);
}

int weightedRandom(vector<int>& weights) {
    vector<int> prefix(weights.size());
    prefix[0] = weights[0];
    for (int i = 1; i < (int)weights.size(); i++)
        prefix[i] = prefix[i-1] + weights[i];
    int r = rand() % prefix.back() + 1;
    return lower_bound(prefix.begin(), prefix.end(), r) - prefix.begin();
}

int rand10() {

vector<int> randomSelect(int M, int N) {
    vector<int> arr(M);
    iota(arr.begin(), arr.end(), 0);       // {0, 1, ..., M-1}
    for (int i = 0; i < N; i++) {
        int j = i + rand() % (M - i);     // 在 [i, M-1] 中随机选一个
        swap(arr[i], arr[j]);
    }
    return vector<int>(arr.begin(), arr.begin() + N);
}

15. 0-1 背包问题（口述 + 手写）#

int knapsack(vector<int>& w, vector<int>& v, int W) {
    int n = w.size();
    // dp[i][j] = 前 i 个物品、容量 j 的最大价值
    vector<vector<int>> dp(n + 1, vector<int>(W + 1, 0));
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= W; j++) {
            dp[i][j] = dp[i-1][j];                    // 不选第 i 个
            if (j >= w[i-1])
                dp[i][j] = max(dp[i][j], dp[i-1][j - w[i-1]] + v[i-1]);  // 选
        }
    }
    return dp[n][W];
}

空间优化（一维滚动数组——面试加分）：

int knapsack(vector<int>& w, vector<int>& v, int W) {
    int n = w.size();
    vector<int> dp(W + 1, 0);
    for (int i = 0; i < n; i++)
        for (int j = W; j >= w[i]; j--)         // 逆序遍历！保证每个物品只用一次
            dp[j] = max(dp[j], dp[j - w[i]] + v[i]);
    return dp[W];
}

复杂度： O(n × W) 时间，优化后 O(W) 空间。

关键追问：

  • 为什么逆序遍历？ 正序会导致同一物品被多次选取（变成完全背包）。
  • 完全背包怎么写？ 正序遍历即可。
  • 游戏应用： 装备选择优化、技能加点方案、背包系统容量管理。
